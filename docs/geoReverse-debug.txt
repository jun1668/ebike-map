const cloud = require('wx-server-sdk'); const https = require('https'); cloud.init({ env: cloud.DYNAMIC_CURRENT_ENV }); const KEY = process.env.TENCENT_MAP_KEY || 'PUT_YOUR_KEY_HERE'; function get(url) { return new Promise(function (resolve) { const req = https.get(url, { timeout: 8000 }, function (res) { let raw = ''; res.on('data', function (c) { raw += c }); res.on('end', function () { resolve(raw) }) }); req.on('error', function (e) { resolve('ERR:' + e.message) }); req.on('timeout', function () { req.destroy(); resolve('TIMEOUT') }) }) } exports.main = async function (event) { const lat = Number(event.latitude); const lng = Number(event.longitude); if (!isFinite(lat) || !isFinite(lng)) { return { ok: true, data: {} } } const url = 'https://apis.map.qq.com/ws/geocoder/v1/?location=' + lat + ',' + lng + '&key=' + KEY + '&get_poi=0'; const raw = await get(url); let j = null; try { j = JSON.parse(raw) } catch (e) { return { ok: true, data: { configured: true, debug: 'PARSE_FAIL', raw: String(raw).slice(0, 200) } } } if (!j || j.status !== 0) { return { ok: true, data: { configured: true, city: '', debug: 'API_ERROR', status: j && j.status, message: j && j.message } } } const ad = j.result.ad_info || {}; return { ok: true, data: { configured: true, province: ad.province, city: ad.city, district: ad.district, address: j.result.address } } }
