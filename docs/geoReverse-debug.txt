const cloud = require('wx-server-sdk'); const https = require('https'); cloud.init({ env: cloud.DYNAMIC_CURRENT_ENV }); const KEY = process.env.TENCENT_MAP_KEY || 'PUT_YOUR_KEY_HERE'; const CACHE_TTL = 6 * 60 * 60 * 1000; const CACHE_MAX = 500; const cache = new Map(); function cacheKey(lat, lng) { return lat.toFixed(2) + ',' + lng.toFixed(2) } function cacheGet(k) { const hit = cache.get(k); if (!hit) { return null } if (Date.now() - hit.at > CACHE_TTL) { cache.delete(k); return null } return hit.data } function cacheSet(k, data) { if (cache.size >= CACHE_MAX) { const first = cache.keys().next().value; if (first !== undefined) { cache.delete(first) } } cache.set(k, { at: Date.now(), data: data }) } function ok(data) { return { ok: true, data: data } } function request(url) { return new Promise(function (resolve, reject) { const req = https.get(url, { timeout: 6000 }, function (res) { let raw = ''; res.on('data', function (chunk) { raw += chunk }); res.on('end', function () { try { resolve(JSON.parse(raw)) } catch (e) { reject(e) } }) }); req.on('timeout', function () { req.destroy(new Error('timeout')) }); req.on('error', reject) }) } exports.main = async function (event) { const lat = Number(event && event.latitude); const lng = Number(event && event.longitude); if (!isFinite(lat) || !isFinite(lng)) { return ok({}) } if (!KEY) { return ok({ configured: false }) } const ck = cacheKey(lat, lng); const cached = cacheGet(ck); if (cached) { return ok(Object.assign({ cached: true }, cached)) } try { const url = 'https://apis.map.qq.com/ws/geocoder/v1/?location=' + lat + ',' + lng + '&key=' + KEY + '&get_poi=0'; const data = await request(url); if (!data || data.status !== 0) { const status = data && data.status; const message = (data && data.message) || ''; console.error('[geoReverse] api error', status, message); return ok({ configured: true, city: '', debug: 'API_ERROR', status: status, message: message }) } const comp = (data.result && data.result.address_component) || {}; const payload = { configured: true, province: comp.province || '', city: comp.city || '', district: comp.district || '', address: (data.result && data.result.address) || '' }; if (payload.city) { cacheSet(ck, payload) } return ok(payload) } catch (err) { console.error('[geoReverse] error', err && err.message); return ok({ configured: true, city: '', debug: 'REQUEST_FAIL', message: String((err && err.message) || err) }) } }
